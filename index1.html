        const AudioContext = window.AudioContext || window.webkitAudioContext;

        let audioCtx;


        function initAudio() {

            if (!audioCtx) audioCtx = new AudioContext();

            if (audioCtx.state === 'suspended') audioCtx.resume();

        }


        function playCorrectSound() {

            try {

                initAudio();

                const now = audioCtx.currentTime;

                const osc1 = audioCtx.createOscillator();

                const gain1 = audioCtx.createGain();

                osc1.type = 'sine';

                osc1.frequency.setValueAtTime(523.25, now); // C5

                gain1.gain.setValueAtTime(0.3, now);

                gain1.gain.exponentialRampToValueAtTime(0.01, now + 0.3);

                osc1.connect(gain1);

                gain1.connect(audioCtx.destination);

                osc1.start(now);

                osc1.stop(now + 0.3);


                const osc2 = audioCtx.createOscillator();

                const gain2 = audioCtx.createGain();

                osc2.type = 'sine';

                osc2.frequency.setValueAtTime(659.25, now + 0.12); // E5

                gain2.gain.setValueAtTime(0.3, now + 0.12);

                gain2.gain.exponentialRampToValueAtTime(0.01, now + 0.45);

                osc2.connect(gain2);

                gain2.connect(audioCtx.destination);

                osc2.start(now + 0.12);

                osc2.stop(now + 0.45);

            } catch(e) {}

        }


        function playWrongSound() {

            try {

                const now = audioCtx.currentTime;

                const osc = audioCtx.createOscillator();

                const gain = audioCtx.createGain();

                osc.type = 'sawtooth';

                osc.frequency.setValueAtTime(180, now);

                gain.gain.setValueAtTime(0.3, now);

                osc.connect(gain);

                gain.connect(audioCtx.destination);

                osc.start(now);

                osc.stop(now + 0.35);

            } catch(e) {}

        }

        function generateQuestions() {

            questions = [];

            for (let i = 0; i < TOTAL_QUESTIONS; i++) {

                const a = Math.floor(Math.random() * 11);

                const b = Math.floor(Math.random() * 11);

                const c = a * b;

                const missingType = Math.floor(Math.random() * 3);


                let expectedAnswer;

                if (missingType === 0) expectedAnswer = c;

                else if (missingType === 1) expectedAnswer = a;

                else expectedAnswer = b;


                questions.push({ a, b, c, missingType, expectedAnswer });

            }

        }


        function loadQuestion() {

            if (currentQuestionIndex >= TOTAL_QUESTIONS) {

                endGame();

                return;

            }


            const q = questions[currentQuestionIndex];

            const display = document.getElementById('equation-display');

            document.getElementById('player-info-display').textContent = `${currentClass.toUpperCase()} (${currentStudentNo}號)`;

            document.getElementById('question-progress').textContent = `題目: ${currentQuestionIndex + 1} / ${TOTAL_QUESTIONS}`;

            document.getElementById('score-display').textContent = `得分: ${score}`;

            document.getElementById('feedback-msg').textContent = '';


            if (q.missingType === 0) {

                display.innerHTML = `${q.a} × ${q.b} = <input type="number" id="user-answer" class="answer-input-box" readonly>`;

            } else if (q.missingType === 1) {

                display.innerHTML = `<input type="number" id="user-answer" class="answer-input-box" readonly> × ${q.b} = ${q.c}`;

            } else {

                display.innerHTML = `${q.a} × <input type="number" id="user-answer" class="answer-input-box" readonly> = ${q.c}`;

            }


            startTimer();

        }


        function startTimer() {

            clearInterval(timerInterval);

            const timerBar = document.getElementById('timer-bar');

            timerBar.style.width = '100%';

            timerBar.style.backgroundColor = 'var(--accent-color)';


            const startTime = Date.now();

            const duration = 10000;


            timerInterval = setInterval(() => {

                const elapsed = Date.now() - startTime;

                const remaining = duration - elapsed;

                

                if (remaining <= 0) {

                    clearInterval(timerInterval);

                    timerBar.style.width = '0%';

                    handleTimeout();

                } else {

                    const percentage = (remaining / duration) * 100;

                    timerBar.style.width = percentage + '%';

                    if (percentage < 30) {

                        timerBar.style.backgroundColor = 'var(--danger-color)';

                    }

                }

            }, 100);

        }


        function pressNum(num) {

            const input = document.getElementById('user-answer');

            if (input && input.value.length < 3) {

                input.value += num;

            }

        }


        function pressDelete() {

            const input = document.getElementById('user-answer');

            if (input) input.value = input.value.slice(0, -1);

        }


        function submitAnswer() {

            const input = document.getElementById('user-answer');

            if (!input || input.value === '') return;


            clearInterval(timerInterval);

            const userAns = parseInt(input.value, 10);

            const q = questions[currentQuestionIndex];

            const feedback = document.getElementById('feedback-msg');


            if (userAns === q.expectedAnswer) {

                score++;

                playCorrectSound();

                feedback.style.color = "var(--accent-color)";

                feedback.textContent = "✨ 答對了！加1分！";

            } else {

                playWrongSound();

                feedback.style.color = "var(--danger-color)";

                feedback.textContent = `❌ 答錯了！正確答案是 ${q.expectedAnswer}`;

            }


            disableInputs();

            setTimeout(() => {

                currentQuestionIndex++;

                loadQuestion();

            }, 1200);

        }


        function handleTimeout() {

            playWrongSound();

            const feedback = document.getElementById('feedback-msg');

            const q = questions[currentQuestionIndex];

            feedback.style.color = "var(--danger-color)";

            feedback.textContent = `⏰ 時間到！正確答案是 ${q.expectedAnswer}`;


            disableInputs();

            setTimeout(() => {

                currentQuestionIndex++;

                loadQuestion();

            }, 1200);

        }


        function disableInputs() {

            const btns = document.querySelectorAll('.numpad button');

            btns.forEach(b => b.disabled = true);

            setTimeout(() => {

                btns.forEach(b => b.disabled = false);

            }, 1200);

        }


        document.addEventListener('keydown', function(e) {

            const gameScreen = document.getElementById('game-screen');

            if (!gameScreen.classList.contains('active')) return;


            if (e.key >= '0' && e.key <= '9') pressNum(parseInt(e.key));

            else if (e.key === 'Backspace') pressDelete();

            else if (e.key === 'Enter') submitAnswer();

        });


        /* ==========================================

           4. 結算與排行榜

           ========================================== */

        function endGame() {

            clearInterval(timerInterval);

            

            document.getElementById('summary-player').textContent = `班別：${currentClass.toUpperCase()}  |  學號：${currentStudentNo} 號`;

            document.getElementById('summary-score').textContent = `${score} / 10`;


            let evalText = "";

            if (score === 10) evalText = "🌟 太厲害了！你是乘法小達人！ 🌟";

            else if (score >= 7) evalText = "👍 做得好！繼續保持！";

            else if (score >= 5) evalText = "💪 還不錯，再多練習一下會更好！";

            else evalText = "📖 不要灰心，每天練習一定會進步！";


            document.getElementById('summary-eval').textContent = evalText;

            document.getElementById('summary-modal').classList.add('active');


            saveScore(currentClass.toUpperCase(), currentStudentNo, score);

        }


        function restartGame() {

            document.getElementById('summary-modal').classList.remove('active');

            startGame();

        }


        function closeModalAndShowLeaderboard() {

            document.getElementById('summary-modal').classList.remove('active');

            showLeaderboardScreen();

        }


        function saveScore(cls, no, scoreVal) {

            const record = {

                class: cls,

                studentNo: no,

                score: scoreVal,

                timestamp: Date.now()

            };


            if (isFirebaseActive) {

                firebase.database().ref('leaderboard').push(record);

            } else {

                let localData = JSON.parse(localStorage.getItem('multiplication_scores') || '[]');

                localData.push(record);

                localStorage.setItem('multiplication_scores', JSON.stringify(localData));

            }

        }


        function showLeaderboardScreen() {

            showScreen('leaderboard-screen');

            const tbody = document.getElementById('leaderboard-body');

            tbody.innerHTML = '<tr><td colspan="3">載入中...</td></tr>';


            if (isFirebaseActive) {

                firebase.database().ref('leaderboard').orderByChild('score').limitToLast(20).once('value')

                    .then(snapshot => {

                        let data = [];

                        snapshot.forEach(child => {

                            data.push(child.val());

                        });

                        data.reverse();

                        renderLeaderboard(data);

                    })

                    .catch(() => {

                        loadLocalLeaderboard();

                    });

            } else {

                loadLocalLeaderboard();

            }

        }


        function loadLocalLeaderboard() {

            let localData = JSON.parse(localStorage.getItem('multiplication_scores') || '[]');

            localData.sort((a, b) => b.score - a.score || b.timestamp - a.timestamp);

            renderLeaderboard(localData.slice(0, 20));

        }


        function renderLeaderboard(list) {

            const tbody = document.getElementById('leaderboard-body');

            tbody.innerHTML = '';


            if (list.length === 0) {

                tbody.innerHTML = '<tr><td colspan="3">暫無紀錄，快來成為第一個挑戰者吧！</td></tr>';

                return;

            }


            list.forEach((item, index) => {

                const tr = document.createElement('tr');

                let rankIcon = index + 1;

                if (index === 0) rankIcon = '🥇 1';

                else if (index === 1) rankIcon = '🥈 2';

                else if (index === 2) rankIcon = '🥉 3';


                tr.innerHTML = `

                    <td><strong>${rankIcon}</strong></td>

                    <td>${item.class} 班 (${item.studentNo}號)</td>

                    <td><strong style="color:var(--primary-color);">${item.score} 分</strong></td>

                `;

                tbody.appendChild(tr);

            });

        }
